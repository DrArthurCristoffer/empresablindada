        function enviarParaWhatsApp() {
            var nome = document.getElementById('nomeLead').value.trim();
            var telefone = document.getElementById('whatsLead').value.trim();
            var interesse = document.getElementById('interesseLead').value;

            if (nome === "" || telefone === "") {
                alert("Por favor, preencha seu nome e WhatsApp para continuarmos de forma segura.");
                return;
            }

            var seuNumeroWhatsApp = "5531971466764"; 
            var texto = "Olá, Dr. Arthur! Meu nome é *" + nome + "*. Preenchi o formulário na sua página e tenho interesse na área de *" + interesse + "*. Gostaria de entender melhor como funciona o diagnóstico preliminar.";
            
            var url = "https://wa.me/" + seuNumeroWhatsApp + "?text=" + encodeURIComponent(texto);
            window.open(url, '_blank');
        }

        function enviarNewsletter(event) {
            event.preventDefault();
            
            const btn = document.getElementById('btnNewsletter');
            const msg = document.getElementById('msgNewsletter');
            const form = document.getElementById('formNewsletter');
            
            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin mr-2"></i> Processando...';
            btn.disabled = true;

            // Coleta todas as áreas marcadas nas caixinhas (Trabalhista, Tributário...)
            const areasCheckbox = document.querySelectorAll('input[name="newsArea"]:checked');
            let areasSelecionadas = [];
            areasCheckbox.forEach((cb) => { areasSelecionadas.push(cb.value); });
            const areaFinal = areasSelecionadas.length > 0 ? areasSelecionadas.join(", ") : "Geral";

            const formData = new FormData();
            formData.append('nome', document.getElementById('newsNome').value);
            formData.append('email', document.getElementById('newsEmail').value);
            formData.append('areas', areaFinal);

            const scriptURL = 'https://script.google.com/macros/s/AKfycbzv3nbX6g6CFpdZkQ3LJWudjtR6QGfkQvqYmsraGaWpwO8xt4KYsqZ4sty8HIe9u0KL/exec';

            fetch(scriptURL, { method: 'POST', body: formData, mode: 'no-cors' })
                .then(() => {
                    msg.innerHTML = '<i class="fa-solid fa-circle-check mr-2"></i> Inscrição confirmada com sucesso!';
                    msg.classList.remove('hidden', 'text-red-400');
                    msg.classList.add('text-green-400');
                    form.reset();
                    btn.innerHTML = '<i class="fa-solid fa-paper-plane mr-2"></i> Quero receber o Boletim';
                    btn.disabled = false;
                })
                .catch(error => {
                    msg.innerHTML = '<i class="fa-solid fa-circle-xmark mr-2"></i> Ocorreu um erro na conexão. Tente novamente.';
                    msg.classList.remove('hidden', 'text-green-400');
                    msg.classList.add('text-red-400');
                    btn.innerHTML = 'Tentar novamente';
                    btn.disabled = false;
                });
        }
    </script>
